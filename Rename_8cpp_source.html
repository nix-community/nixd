91const Inherit *upInherit(const Node &Name, const ParentMapAnalysis &PMA) {
92 // The expression of `inherit (E) a;` is a child of the inherit too.
93 if (Name.kind() != Node::NK_AttrName)
94 return nullptr;
95 const Node *Up = PMA.query(Name);
96 if (Up && Up->kind() == Node::NK_Inherit)
97 return static_cast<const Inherit *>(Up);
98 return nullptr;
99}

101/// A name in `inherit` is both the attribute's name and the variable (or the
102/// attribute of the inherited-from expression), so renaming one of them
103/// rewrites it as its own binding \p Binding, keeping the inherit's other
104/// names.
105void splitInherit(const Inherit &Inh, const Node &Name, std::string Binding,
106 llvm::StringRef Src,
107 std::vector<lspserver::TextEdit> &Edits) {
108 const auto &Names = Inh.names();
109 if (Names.size() == 1) {
110 Edits.emplace_back(lspserver::TextEdit{
111 .range = toLSPRange(Src, Inh.range()),
112 .newText = std::move(Binding),
113 });
114 return;
115 }
116 // The name and the spaces before it on its line: comments stay.
117 LexerCursorRange Removal = withLeadingSpaces(Name, Src);
118 // Names need no space between them when one is quoted: removing `"b"` from
119 // `a"b"c` must not join `a` and `c` into `ac`.
120 Edits.emplace_back(lspserver::TextEdit{
121 .range = toLSPRange(Src, Removal),
122 .newText = gluesNeighbours(Removal, Src) ? " " : "",
123 });
124 Edits.emplace_back(lspserver::TextEdit{
125 .range = toLSPRange(Src, LexerCursorRange{Inh.rCur()}),
126 .newText = " " + std::move(Binding),
127 });
128}

145/// `inherit (E) name;` selects `name` from `E`, whose text is \p From: the
146/// source of `E` with the renamed uses inside it already replaced.
147std::string selectFrom(const Expr &E, const std::string &From,
148 llvm::StringRef Name) {
149 bool Bare = E.kind() == Node::NK_ExprVar ||
150 (E.kind() == Node::NK_ExprSelect &&
151 !static_cast<const ExprSelect &>(E).defaultExpr());
152 return (Bare ? From : "(" + From + ")") + "." + Name.str();
153}

155/// The source of \p E, with each of \p Uses (all inside it) renamed.
156std::string renamedSource(const Expr &E, std::vector<const ExprVar *> Uses,
157 const std::string &NewText, llvm::StringRef Src) {
158 std::string Text(E.src(Src));
159 std::sort(Uses.begin(), Uses.end(), [](const auto *A, const auto *B) {
160 return A->lCur().offset() > B->lCur().offset();
161 });
162 for (const ExprVar *Use : Uses) {
163 auto Begin = Use->lCur().offset() - E.lCur().offset();
164 Text.replace(Begin, Use->rCur().offset() - Use->lCur().offset(), NewText);
165 }
166 return Text;
167}

203/// The environment \p D, named \p Name, is defined in.
204const EnvNode *defEnv(const Definition &D, const std::string &Name,
205 const ParentMapAnalysis &PMA,
206 const VariableLookupAnalysis &VLA) {
207 auto Up = [&](const EnvNode *E) -> const EnvNode * {
208 for (; E; E = E->parent())
209 if (defines(*E, Name, D))
210 return E;
211 return nullptr;
212 };
213 for (const ExprVar *Use : D.uses())
214 if (const EnvNode *E = Up(VLA.env(Use)))
215 return E;
216 // Without uses, the environment of the scope's body.
217 const Node *Body = nullptr;
218 switch (D.source()) {
219 case Definition::DS_Let:
220 if (const Node *Attrs = PMA.upTo(*D.syntax(), Node::NK_ExprAttrs))
221 if (const Node *Let = PMA.query(*Attrs);
222 Let && Let->kind() == Node::NK_ExprLet)
223 Body = static_cast<const ExprLet &>(*Let).expr();
224 break;
225 case Definition::DS_LambdaArg:
226 case Definition::DS_LambdaNoArg_Formal:
227 case Definition::DS_LambdaWithArg_Arg:
228 case Definition::DS_LambdaWithArg_Formal:
229 case Definition::DS_FlakeInjectedFormal:
230 if (const Node *Lambda = PMA.upTo(*D.syntax(), Node::NK_ExprLambda))
231 Body = static_cast<const ExprLambda &>(*Lambda).body();
232 break;
233 default:
234 break;
235 }
236 return Body ? Up(VLA.env(Body)) : nullptr;
237}

239/// Calls \p F on every variable, those that `inherit name;` desugars into
240/// included.
241void forEachVar(const Node &N, const std::function<void(const ExprVar &)> &F) {
242 if (N.kind() == Node::NK_ExprVar)
243 F(static_cast<const ExprVar &>(N));
244 if (N.kind() == Node::NK_ExprAttrs)
245 for (const auto &[_, Attr] :
246 static_cast<const ExprAttrs &>(N).sema().staticAttrs())
247 if (Attr.kind() == Attribute::AttributeKind::Inherit && Attr.value() &&
248 Attr.value()->kind() == Node::NK_ExprVar)
249 F(static_cast<const ExprVar &>(*Attr.value()));
250 for (const Node *Child : N.children())
251 if (Child)
252 forEachVar(*Child, F);
253}

255/// Refuses a rename of \p D from \p Old to \p New that would change what a
256/// variable reads: a binding of \p New between a use and \p D would capture
257/// the use, and the renamed binding would capture a use of \p New that reads
258/// something outside it.
259void checkCapture(const Definition &D, const std::string &Old,
260 const std::string &New, const Node &Root,
261 const ParentMapAnalysis &PMA,
262 const VariableLookupAnalysis &VLA) {
263 if (New.empty())
264 return;
265 // A variable is named by an identifier: `"y"` would bind `y` but be written
266 // as a string where the variable is used.
267 if (!isValidNixIdentifier(New))
268 throw RenameCaptureException("cannot rename: `" + New +
269 "` is not a valid identifier");
270 // `__curPos` is the position it is written at, never a variable.
271 if (New == "__curPos")
272 throw RenameCaptureException(
273 "cannot rename: `__curPos` is the current position, not a variable");
274 const EnvNode *DefE = defEnv(D, Old, PMA, VLA);
275 if (DefE && binds(*DefE, New))
276 throw RenameCaptureException(
277 "cannot rename: " + describe(*DefE->defs().at(New), New) +
278 " is in the same scope");
279
280 for (const ExprVar *Use : D.uses())
281 for (const EnvNode *E = VLA.env(Use); E && !defines(*E, Old, D);
282 E = E->parent())
283 if (binds(*E, New))
284 throw RenameCaptureException(
285 "cannot rename: the use of `" + Old + "` at " + at(*Use) +
286 " would read " + describe(*E->defs().at(New), New) + " instead");
287
288 if (!DefE)
289 return;
290 forEachVar(Root, [&](const ExprVar &Var) {
291 if (Var.id().name() != New)
292 return;
293 for (const EnvNode *E = VLA.env(&Var); E; E = E->parent()) {
294 if (binds(*E, New))
295 return;
296 if (E != DefE)
297 continue;
298 auto R = VLA.query(Var);
299 std::string Reads =
300 R.Def ? describe(*R.Def, New) : "`" + New + "`, which is undefined";
301 throw RenameCaptureException(
302 "cannot rename: the use of `" + New + "` at " + at(Var) + " reads " +
303 Reads + ", but would read the renamed `" + Old + "`");
304 }
305 });
306}

308WorkspaceEdit rename(const nixf::Node &Desc, const std::string &NewText,
309 const ParentMapAnalysis &PMA,
310 const VariableLookupAnalysis &VLA, const URIForFile &URI,
311 llvm::StringRef Src) {
312 using lspserver::TextEdit;
313 // Find "definition"
314 const Definition &Def = findDefinition(Desc, PMA, VLA);
315
316 if (Def.source() == Definition::DS_With)
317 throw RenameWithException();
318
319 if (Def.isBuiltin())
320 throw RenameBuiltinException();
321
322 if (Def.source() == Definition::DS_Rec)
323 throw RenameRecException();
324
325 // The expression `__curPos` is the position it is written at, however a
326 // binding of that name is spelled (`inherit "__curPos";` too), so what
327 // looks like its uses are not, and a rename would change them.
328 if (definedName(*Def.syntax(), Src) == "__curPos")
329 throw RenameCurPosException();
330
331 // Renaming to the current name changes nothing, but splitting `inherit x;`
332 // would write `x = x;`, which recurses.
333 if (definedName(*Def.syntax(), Src) == NewText) {
334 WorkspaceEdit WE;
335 WE.changes = std::map<std::string, std::vector<TextEdit>>{{URI.uri(), {}}};
336 return WE;
337 }
338
339 const Node *Root = &Desc;
340 while (!PMA.isRoot(*Root) && PMA.query(*Root))
341 Root = PMA.query(*Root);
342
343 checkCapture(Def, definedName(*Def.syntax(), Src), NewText, *Root, PMA, VLA);
344
345 std::vector<TextEdit> Edits;
346
347 // A definition from `inherit (E) name;` is rewritten as a binding that
348 // copies `E`, so uses inside `E` are renamed in that copy.
349 const Inherit *DefInh = upInherit(*Def.syntax(), PMA);
350 const Expr *From = DefInh ? DefInh->expr().get() : nullptr;
351 if (From && DefInh->names().size() > 1 && !isShared(*From))
352 throw RenameInheritSharedException();
353 std::vector<const ExprVar *> UsesInFrom;
354
355 // `inherit (E) a b;` desugars into one selection per name, all sharing `E`,
356 // so a use inside `E` is recorded once per name.
357 std::set<const ExprVar *> Seen;
358 for (const auto *Use : Def.uses()) {
359 if (!Seen.insert(Use).second)
360 continue;
361 // `inherit name;` desugars into a variable sharing the name's identifier.
362 const Node *Name = PMA.query(Use->id());
363 const Inherit *Inh = Name ? upInherit(*Name, PMA) : nullptr;
364 if (From && contains(*From, *Use)) {
365 if (Inh)
366 throw RenameInheritNestedException();
367 UsesInFrom.emplace_back(Use);
368 // A single-name inherit is replaced whole, `E` included.
369 if (DefInh->names().size() == 1)
370 continue;
371 } else if (Inh) {
372 const Node *Attrs = PMA.upTo(*Inh, Node::NK_ExprAttrs);
373 if (Attrs) {
374 const SemaAttrs &SA = static_cast<const ExprAttrs &>(*Attrs).sema();
375 if (SA.isRecursive() && SA.staticAttrs().contains(NewText))
376 throw RenameInheritCaptureException();
377 }
378 splitInherit(*Inh, *Name,
379 std::string(Name->src(Src)) + " = " + NewText + ";", Src,
380 Edits);
381 continue;
382 }
383 Edits.emplace_back(TextEdit{
384 .range = toLSPRange(Src, Use->range()),
385 .newText = NewText,
386 });
387 }
388
389 if (DefInh) {
390 std::string Name(Def.syntax()->src(Src));
391 if (From && mentionsCurPos(*From))
392 throw RenameInheritMovesCurPosException();
393 std::string Value =
394 From ? selectFrom(*From, renamedSource(*From, UsesInFrom, NewText, Src),
395 Name)
396 : Name;
397 splitInherit(*DefInh, *Def.syntax(), NewText + " = " + Value + ";", Src,
398 Edits);
399 } else {
400 Edits.emplace_back(TextEdit{
401 .range = toLSPRange(Src, Def.syntax()->range()),
402 .newText = NewText,
403 });
404 }
405 WorkspaceEdit WE;
406 WE.changes = std::map<std::string, std::vector<TextEdit>>{
407 {URI.uri(), std::move(Edits)}};
408 return WE;
409}

412void Controller::onRename(const RenameParams &Params,
413 Callback<WorkspaceEdit> Reply) {
414 using CheckTy = WorkspaceEdit;
415 auto Action = [Reply = std::move(Reply), URI = Params.textDocument.uri,
416 Pos = toNixfPosition(Params.position),
417 NewText = Params.newName, this]() mutable {
418 const auto File = URI.file();
419 return Reply([&]() -> llvm::Expected<WorkspaceEdit> {
420 const auto TU = CheckDefault(getTU(File));
421 const auto AST = CheckDefault(getAST(*TU));
422 const auto &Desc = *CheckReturn(
423 AST->descend({Pos, Pos}),
424 error("cannot find corresponding node on given position"));
425
426 const auto &PM = *TU->parentMap();
427 const auto &VLA = *TU->variableLookup();
428 try {
429 return rename(Desc, NewText, PM, VLA, URI, TU->src());
430 } catch (std::exception &E) {
431 return error(E.what());
432 }
433 }());
434 };
435 boost::asio::post(Pool, std::move(Action));
436}
