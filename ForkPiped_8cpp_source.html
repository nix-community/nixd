42 auto &PipeErr = Pipes.Err;
43 if (Platform.createPipe(PipeIn) == -1 || Platform.createPipe(PipeOut) == -1 ||

45 throw std::system_error(errno, std::generic_category());
46
47 // Keep the sources away from stdio, even if the caller closed fd 0, 1 or 2.
48 // This also ensures dup2 clears CLOEXEC rather than becoming a no-op.
49 for (auto *Pipe : {PipeIn, PipeOut, PipeErr})
50 for (int I = 0; I < 2; ++I)
51 if (Pipe[I] <= STDERR_FILENO) {
52 int FD = fcntl(Pipe[I], F_DUPFD_CLOEXEC, STDERR_FILENO + 1);
53 if (FD == -1)
54 throw std::system_error(errno, std::generic_category());
55 close(Pipe[I]);
56 Pipe[I] = FD;
57 }
58
59 pid_t Child = Platform.forkProcess();
60 if (Child == -1)
61 throw std::system_error(errno, std::generic_category());
62
63 if (Child == 0) {
64 // Redirect stdin, stdout, stderr.
65 const int Sources[] = {PipeIn[READ], PipeOut[WRITE], PipeErr[WRITE]};
66 for (int Target = STDIN_FILENO; Target <= STDERR_FILENO; ++Target) {
67 int Result;
68 do {
69 Result = dup2(Sources[Target], Target);
70 } while (Result == -1 && errno == EINTR);
71 if (Result == -1)
72 _exit(127);
73 }
74 return 0;
75 }
76
77 In = std::exchange(PipeIn[WRITE], -1);
78 Out = std::exchange(PipeOut[READ], -1);
79 Err = std::exchange(PipeErr[READ], -1);
80 return Child;
